1. 使用启动脚本运行docker
bash run_marvin_m6_impedance.sh

/usr/bin/python3 -m pip install pyserial minimalmodbus modbus_tk 
unset ROS_LOCALHOST_ONLY
export ROS_DOMAIN_ID=42
source /marvin_ws/install/setup.bash


2. 启动大臂和夹爪
(1) 不启动夹爪：
ros2 launch robot_driver robot_servo_start_marvin.launch.py ip:=6.6.7.190
(2) 启动夹爪：
changing夹爪启动
ros2 launch robot_driver robot_servo_start_marvin.launch.py \
  ip:=6.6.7.190 \
  use_left_gripper:=true \
  use_right_gripper:=true \
  gripper_type:=changingtek \
  gripper_port_left:=/dev/ttyUSB0 \
  gripper_port_right:=/dev/ttyUSB1
robotiq夹爪启动
ros2 launch robot_driver robot_servo_start_marvin.launch.py \
  ip:=6.6.7.190 \
  use_left_gripper:=true \
  use_right_gripper:=true \
  gripper_type:=robotiq \
  gripper_port_left:=/dev/ttyUSB0 \
  gripper_port_right:=/dev/ttyUSB1
  
阻抗模式/位置模式参数设置 /home/wsj/pycode/marvin_ws_humble_add_impedance/marvin_ws/install/share/robot_driver/config/robot_params_marvin_m6.yaml
